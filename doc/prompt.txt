O código abaixo está funcionando, porém não é viável a longo prazo, quando o número de notas for muito grande.
Preciso de uma solução em que vou digitando o título da nota e a lista vai reduzindo.

// -------- LINK INTERNO --------
const refs = await listNoteReferences();

const options = refs.data
    .map((n, i) => `${i + 1} - ${n.titulo}`)
    .join("\n");

const choice = prompt(
    "Escolha a nota de destino:\n\n" + options
);

if (!choice) return;

const index = Number(choice) - 1;
const notaDestino = refs.data[index];

if (!notaDestino) {
    alert("Opção inválida");
    return;
}

currentLinks.push({
    tipo: "interno",
    nota_destino_id: notaDestino.id,
    nota_destino_titulo: notaDestino.titulo
});

renderLinksList(currentLinks);
